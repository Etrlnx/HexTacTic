graph LR
    classDef input fill:#2B6CB0,stroke:#2C5282,color:#FFF;
    classDef preproc fill:#319795,stroke:#234E52,color:#FFF;
    classDef nn fill:#805AD5,stroke:#6B46C1,color:#FFF;
    classDef rl fill:#D69E2E,stroke:#B7791F,color:#FFF;

    subgraph Environment ["Game Environment"]
        RawBoard["Raw Board List (6x6)"]:::input
        ValidMoves["Valid Actions Mask"]:::input
    end

    subgraph Preprocessing ["Tensor Preprocessing (preproc_s)"]
        Ch0["Channel 0: Open Spaces (1/0)"]:::preproc
        Ch1["Channel 1: My Pieces (1/0)"]:::preproc
        Ch2["Channel 2: Opponent Pieces (1/0)"]:::preproc
        TensorStack["Stacked 4D Tensor (1, 3, 6, 6)"]:::preproc
    end

    subgraph ModelArch ["TTDQNN Architecture"]
        Conv1["Conv2d (3 -> 32, 3x3) + ReLU"]:::nn
        Conv2["Conv2d (32 -> 64, 3x3) + ReLU"]:::nn
        Flatten["Flatten (64 * 6 * 6 = 2304)"]:::nn
        FC1["Linear (2304 -> 128) + ReLU"]:::nn
        FC2["Linear Output Layer (128 -> 36 Q-Values)"]:::nn
    end

    subgraph ActionSelection ["Action Selection Logic"]
        QValues["36 Unmasked Q-Values"]:::rl
        Masking["Apply -inf to Illegal Indices"]:::rl
        ArgMax["np.argmax(q_masked)"]:::rl
        SelectedAction["Selected Board Move (0-35)"]:::rl
    end

    subgraph LearningEngine ["DQN Training & Replay Engine"]
        Memory[("Replay Buffer: (S, A, R, S', Done)")]:::rl
        Loss["MSE Loss(Q_val, Target_Q)"]:::rl
        Backprop["Adam Optimizer Step (Updates Weights)"]:::rl
    end
    RawBoard --> Ch0 & Ch1 & Ch2
    Ch0 & Ch1 & Ch2 --> TensorStack
    TensorStack --> Conv1
    Conv1 --> Conv2 --> Flatten --> FC1 --> FC2 --> QValues
    
    QValues --> Masking
    ValidMoves --> Masking
    Masking --> ArgMax --> SelectedAction
    TensorStack & SelectedAction -.->|Store Transitions| Memory
    Memory -.->|Sample Batch of 64| Loss
    Loss -.-> Backprop
